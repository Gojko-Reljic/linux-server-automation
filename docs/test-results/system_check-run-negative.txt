## A. Threshold validation (MIN_RAM_MB, MIN_DISK_GB)

# A1. Before: why validation exists. A threshold taken from the environment and used inside (( )) runs code as root.
$ cat > /tmp/inject_demo.sh <<'EOF'
#!/usr/bin/env bash
MIN_RAM_MB="${MIN_RAM_MB:-512}"
if (( 1024 >= MIN_RAM_MB )); then echo "OK"; fi
EOF
$ sudo rm -f /tmp/pwned
$ sudo env MIN_RAM_MB='a[$(touch /tmp/pwned)]' bash /tmp/inject_demo.sh
OK
$ ls -l /tmp/pwned
-rw-r--r-- 1 root root 0 Oct 15 20:52 /tmp/pwned
# The "threshold" executed touch as root and the script reported OK: the attack is silent.

# A2. After: the same value against the real script.
$ sudo rm -f /tmp/pwned
$ sudo env MIN_RAM_MB='a[$(touch /tmp/pwned)]' bash scripts/system_check.sh;echo "exit kod: $?"
[ERROR] MIN_RAM_MB must be a positive integer, got 'a[$(touch /tmp/pwned)]'.
exit kod: 1
$ ls -l /tmp/pwned
ls: cannot access '/tmp/pwned': No such file or directory
# Rejected before any check runs (not even the "Running system check" line), and nothing was executed.

# A3. Other invalid values, each hitting a different part of ^[1-9][0-9]*$.
$ sudo env MIN_RAM_MB=abc bash scripts/system_check.sh; echo "exit kod: $?"
[ERROR] MIN_RAM_MB must be a positive integer, got 'abc'.
exit kod: 1
$ sudo env MIN_RAM_MB=0 bash scripts/system_check.sh; echo "exit kod: $?"
[ERROR] MIN_RAM_MB must be a positive integer, got '0'.
exit kod: 1
$ sudo env MIN_RAM_MB=08 bash scripts/system_check.sh; echo "exit kod: $?"
[ERROR] MIN_RAM_MB must be a positive integer, got '08'.
exit kod: 1
$ sudo env MIN_DISK_GB=-5 bash scripts/system_check.sh; echo "exit kod: $?"
[ERROR] MIN_DISK_GB must be a positive integer, got '-5'.
exit kod: 1
# 0 and 08 are valid numbers but rejected on purpose (bash would read 08 as octal). MIN_DISK_GB goes through the same loop.

## B. Privileges: EUID 0 alone is not enough

$ bash scripts/system_check.sh; echo "exit kod: $?"
[ERROR] This script must be run as root (use sudo).
exit kod: 1
$ command -v capsh
/usr/sbin/capsh
$ sudo grep CapEff /proc/self/status
CapEff:	000001ffffffffff
$ sudo capsh --drop=cap_net_admin -- -c 'grep CapEff /proc/self/status'
CapEff:	000001ffffffefff

## C. Storage: free space and free inodes

$ sudo env MIN_DISK_GB=999 bash scripts/system_check.sh; echo "exit kod: $?"
[INFO] === Running system check ===
[INFO] Detected supported OS: ubuntu 26.04 (resolute)
[INFO] Runtime: apt-get, systemd and required capabilities: OK
[INFO] Effective RAM: 3398 MB
[INFO] Free space at /: 4 GB, 642874 inodes
[ERROR] Not enough disk space at /. Required: 999 GB, available: 4 GB.
exit kod: 1
# Stops at the first failing mount point (fail fast), so /var and /home are not printed.
# The inode count is 642874 here and 642878 in the happy path: the test files created in /tmp in between use inodes.

# The real disk never runs low on inodes, so df is replaced by a stub that reports 10 free inodes.
# The script is sourced (main does not run because of the BASH_SOURCE guard) and only check_storage is called.
$ cat > /tmp/stub_df.sh <<'EOF'
df() {
    if [[ $1 == -Pi ]]; then
        printf 'Filesystem Inodes IUsed IFree IUse%% Mounted on\n'
        printf '/dev/stub 1000 990 10 99%% /\n'
    else
        command df "$@"
    fi
}
EOF
$ sudo env MIN_DISK_GB=1 bash -c 'source scripts/system_check.sh; source /tmp/stub_df.sh; check_storage'; echo "exit kod: $?"
[INFO] Free space at /: 4 GB, 10 inodes
[ERROR] Not enough free inodes at /.
exit kod: 1
# Space passes (real df -Pk), the inode branch fails on the stubbed value.

## G. Memory: the cgroup limit is respected

[host] $ sc_new ubuntu:22.04 --memory=256m
[host] $ docker cp /tmp/system_check_old.sh sc-test:/tmp/system_check_old.sh
[host] $ docker exec sc-test bash -c 'source /tmp/system_check_old.sh; check_ram'; echo "exit kod: $?"
[INFO] Total RAM: 31980 MB
exit kod: 0
[host] $ docker exec sc-test bash -c 'source /repo/scripts/system_check.sh; check_memory'; echo "exit kod: $?"
[INFO] Effective RAM: 256 MB
[ERROR] Not enough RAM. Required: 512 MB, available: 256 MB.
exit kod: 1
[host] $ docker exec sc-test cat /sys/fs/cgroup/memory.max
268435456
# 268435456 bytes = 256 MB. The old check reported the host's 31980 MB for a container limited to 256 MB and passed.

[host] $ sc_new ubuntu:22.04 --memory=1g
[host] $ docker exec sc-test bash -c 'source /repo/scripts/system_check.sh; check_memory'; echo "exit kod: $?"
[INFO] Effective RAM: 1024 MB
exit kod: 0
# A limit below the host but above the threshold passes.

[host] $ sc_new ubuntu:22.04
[host] $ docker exec sc-test cat /sys/fs/cgroup/memory.max
max
[host] $ docker exec sc-test bash -c 'source /repo/scripts/system_check.sh; check_memory'; echo "exit kod: $?"
[INFO] Effective RAM: 31980 MB
exit kod: 0
# "max" means no limit: it is not used as a number and the host total applies.
