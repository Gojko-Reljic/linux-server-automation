--- B: .ssh is a symlink ------------------------------------------------

$ docker exec create-users-test mkdir /root/ssh-target
$ docker exec create-users-test stat -c '%U:%G %a %n' /root/ssh-target
root:root 755 /root/ssh-target
$ docker exec create-users-test rmdir /home/deployer/.ssh
$ docker exec create-users-test ln -s /root/ssh-target /home/deployer/.ssh
$ docker exec create-users-test ls -ld /home/deployer/.ssh
lrwxrwxrwx. 1 root root 16 Oct 23 21:17 /home/deployer/.ssh -> /root/ssh-target
$ docker exec create-users-test /root/linux-server-automation/scripts/create_users.sh; echo "exit=$?"
[INFO] Managed user 'deployer' already exists.
[ERROR] Refusing to touch /home/deployer/.ssh: it is a symlink.
exit=1
$ docker exec create-users-test stat -c '%U:%G %a %n' /root/ssh-target
root:root 755 /root/ssh-target
# The script refuses to follow the link, and the link target kept its
# root:root 755. Had the script followed it, chown/chmod would have
# turned it into deployer:deployer 700. It stopped at the first
# account, so webadmin was not processed.
# (the trailing "." after the permissions is the SELinux context marker
#  shown by the Fedora host)

--- C: what a locked sudo account can do --------------------------------

# Preparation: the minimal Docker image has the group "sudo" but not the
# sudo package. (apt output omitted)
$ docker exec create-users-test which sudo; echo "exit=$?"
exit=1
$ docker exec create-users-test apt-get update -qq
$ docker exec create-users-test apt-get install -y -qq sudo
(output omitted: Selecting/Unpacking/Setting up sudo 1.9.9-1ubuntu2.7)

$ docker exec create-users-test grep -n '^%sudo' /etc/sudoers
50:%sudo	ALL=(ALL:ALL) ALL
# The rule has no NOPASSWD: members of the group must give a password.

# C1: account still locked
$ docker exec create-users-test passwd -S deployer
deployer L 10/23/2026 0 99999 7 -1
$ docker exec create-users-test su - deployer -c 'id; sudo -n true'; echo "exit=$?"
uid=1000(deployer) gid=1000(deployer) groups=1000(deployer),27(sudo)
sudo: a password is required
exit=1
# deployer is in the sudo group but has no password to give, so sudo
# does not work yet.
$ docker exec create-users-test sudo -l -U deployer; echo "exit=$?"
Matching Defaults entries for deployer on 32623cc0dc49:
    env_reset, mail_badpass, secure_path=/usr/local/sbin\:/usr/local/bin\:/usr/sbin\:/usr/bin\:/sbin\:/bin\:/snap/bin, use_pty

User deployer may run the following commands on 32623cc0dc49:
    (ALL : ALL) ALL
exit=0
$ docker exec create-users-test sudo -l -U webadmin; echo "exit=$?"
User webadmin is not allowed to run sudo on 32623cc0dc49.
exit=0
# The role is correct on both accounts: deployer may run everything,
# webadmin may run nothing.

# C2: the administrator sets a password (test-only value)
$ docker exec create-users-test sh -c "echo 'deployer:TestOnly-123' | chpasswd"
$ docker exec create-users-test passwd -S deployer
deployer P 10/23/2026 0 99999 7 -1
$ docker exec create-users-test su - deployer -c "echo 'TestOnly-123' | sudo -S -p '' id -u"; echo "exit=$?"
0
exit=0
# With a password set, sudo works (id -u prints 0).
