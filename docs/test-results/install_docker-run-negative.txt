## A. [host] Not root

[host] $ docker run --rm --user 1000:1000 --security-opt label=disable \
    -v "$PWD":/repo:ro ubuntu:24.04 \
    bash /repo/scripts/install_docker.sh
[ERROR] This script must be run as root (use sudo).
exit=1
# require_root stops the script before anything else runs.

## B. [host] Root, but systemd is not the active init system

[host] $ docker run --rm --security-opt label=disable \
    -v "$PWD":/repo:ro ubuntu:24.04 \
    bash /repo/scripts/install_docker.sh
[ERROR] systemd is not the active init system (no /run/systemd/system).
exit=1
# require_systemd stops the script before the first apt-get.
