# OS: Ubuntu 26.04.1 LTS (KVM VM), tested 2026-10-04
# IPv6 policy: the script refuses to run when IPV6 is not enabled in
# /etc/default/ufw, unless FIREWALL_ALLOW_IPV6_DISABLED=true is set explicitly.

## Test 1: IPV6=no, no opt-out -> refused

$ sudo sed -i 's/^IPV6=yes/IPV6=no/' /etc/default/ufw
$ sudo ./scripts/configure_firewall.sh; echo "exit=$?"
[ERROR] IPv6 is not enabled in /etc/default/ufw. Set IPV6=yes, or run with FIREWALL_ALLOW_IPV6_DISABLED=true if this host intentionally has no IPv6.
exit=1
$ sudo sed -i 's/^IPV6=no/IPV6=yes/' /etc/default/ufw
$ grep '^IPV6' /etc/default/ufw
IPV6=yes
